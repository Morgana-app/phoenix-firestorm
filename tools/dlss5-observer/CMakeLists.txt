cmake_minimum_required(VERSION 3.20)
project(firestorm_scene_observer LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The observer requires Windows x64")
endif()
set(RESHADE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/reshade-6.8.0/include" CACHE PATH "ReShade 6.8.0 add-on headers")
if(NOT EXISTS "${RESHADE_INCLUDE_DIR}/reshade.hpp")
    message(FATAL_ERROR "Set RESHADE_INCLUDE_DIR to the ReShade 6.8.0 include folder")
endif()
set(GL_HEADERS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/gl-headers" CACHE PATH "Khronos GL glext.h and KHR/khrplatform.h root")
if(NOT EXISTS "${GL_HEADERS_DIR}/GL/glext.h" OR NOT EXISTS "${GL_HEADERS_DIR}/KHR/khrplatform.h")
    message(FATAL_ERROR "Set GL_HEADERS_DIR to the Khronos OpenGL extension headers root")
endif()
add_library(firestorm_scene_observer SHARED observer.cpp copy_probe.cpp gl_snapshot.cpp identity_probe.cpp early_neural.cpp)
target_include_directories(firestorm_scene_observer PRIVATE "${RESHADE_INCLUDE_DIR}" "${GL_HEADERS_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../indra/newview")
target_compile_features(firestorm_scene_observer PRIVATE cxx_std_17)
target_compile_definitions(firestorm_scene_observer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(firestorm_scene_observer PRIVATE opengl32 user32)
set_target_properties(firestorm_scene_observer PROPERTIES PREFIX "" SUFFIX ".addon64")
if(MSVC)
    target_compile_options(firestorm_scene_observer PRIVATE /W4 /WX /EHsc)
endif()
enable_testing()
add_executable(resize_policy_tests resize_policy_tests.cpp)
target_compile_features(resize_policy_tests PRIVATE cxx_std_17)
add_test(NAME early_neural_resize_policy COMMAND resize_policy_tests)
add_executable(feeder_helper_restart_tests feeder_helper_restart_tests.cpp)
target_compile_features(feeder_helper_restart_tests PRIVATE cxx_std_17)
add_test(NAME feeder_helper_restart_policy COMMAND feeder_helper_restart_tests)
add_test(NAME feeder_helper_restart_opt_out COMMAND feeder_helper_restart_tests --disabled)
get_filename_component(RESHADE_SOURCE_DIR "${RESHADE_INCLUDE_DIR}/../source" ABSOLUTE)
add_executable(identity_shader_tests identity_shader_tests.cpp
    "${RESHADE_SOURCE_DIR}/effect_codegen_glsl.cpp"
    "${RESHADE_SOURCE_DIR}/effect_expression.cpp"
    "${RESHADE_SOURCE_DIR}/effect_lexer.cpp"
    "${RESHADE_SOURCE_DIR}/effect_parser_exp.cpp"
    "${RESHADE_SOURCE_DIR}/effect_parser_stmt.cpp"
    "${RESHADE_SOURCE_DIR}/effect_preprocessor.cpp"
    "${RESHADE_SOURCE_DIR}/effect_symbol_table.cpp")
target_include_directories(identity_shader_tests PRIVATE "${RESHADE_SOURCE_DIR}")
target_compile_features(identity_shader_tests PRIVATE cxx_std_17)
set(IDENTITY_SHADER_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build-vc/newview/Release/reshade-shaders/Shaders" CACHE PATH "Folder containing ReShade.fxh for shader parser tests")
target_compile_definitions(identity_shader_tests PRIVATE NOMINMAX
    IDENTITY_SHADER_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/FS_P1Identity.fx"
    IDENTITY_SHADER_INCLUDE="${IDENTITY_SHADER_INCLUDE_DIR}")
add_test(NAME identity_effect_glsl COMMAND identity_shader_tests)
add_executable(copy_probe_tests copy_probe_tests.cpp copy_probe.cpp gl_snapshot.cpp)
target_include_directories(copy_probe_tests PRIVATE ../../indra/newview "${GL_HEADERS_DIR}")
target_compile_features(copy_probe_tests PRIVATE cxx_std_17)
target_compile_definitions(copy_probe_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX FS_PROBE_TEST_HOOKS)
target_link_libraries(copy_probe_tests PRIVATE opengl32 user32 gdi32)
if(MSVC)
    target_compile_options(copy_probe_tests PRIVATE /W4 /WX /EHsc)
endif()
add_test(NAME scene_copy_native_gl COMMAND copy_probe_tests)
set_tests_properties(scene_copy_native_gl PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME scene_copy_compat_gl COMMAND copy_probe_tests --compat)
set_tests_properties(scene_copy_compat_gl PROPERTIES SKIP_RETURN_CODE 77)
add_executable(observer_tests observer_tests.cpp)
target_compile_features(observer_tests PRIVATE cxx_std_17)
target_compile_definitions(observer_tests PRIVATE OBSERVER_SOURCE_ROOT="${CMAKE_SOURCE_DIR}/../..")
add_test(NAME observer_detection COMMAND observer_tests)
add_executable(scene_signal_c_layout scene_signal_c_layout.c)
target_include_directories(scene_signal_c_layout PRIVATE ../../indra/newview)
add_test(NAME scene_signal_c_abi COMMAND scene_signal_c_layout)

# Exercise the real optional Windows dispatcher without viewer third-party deps.
add_executable(scene_signal_tests scene_signal_tests.cpp ../../indra/newview/fsscenesignal.cpp)
target_include_directories(scene_signal_tests PRIVATE ../../indra/newview)
target_compile_features(scene_signal_tests PRIVATE cxx_std_17)
target_compile_definitions(scene_signal_tests PRIVATE FS_SCENE_SIGNAL_STANDALONE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(scene_signal_test_receiver SHARED scene_signal_test_receiver.cpp)
target_include_directories(scene_signal_test_receiver PRIVATE ../../indra/newview)
target_compile_features(scene_signal_test_receiver PRIVATE cxx_std_17)
set_target_properties(scene_signal_test_receiver PROPERTIES PREFIX "" OUTPUT_NAME firestorm_scene_observer SUFFIX ".addon64"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-receiver"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-receiver")
add_dependencies(scene_signal_tests scene_signal_test_receiver)
add_library(scene_signal_missing_export SHARED scene_signal_missing_export.cpp)
set_target_properties(scene_signal_missing_export PROPERTIES PREFIX "" OUTPUT_NAME firestorm_scene_observer SUFFIX ".addon64"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-missing-export"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-missing-export")
add_dependencies(scene_signal_tests scene_signal_missing_export)
add_test(NAME scene_signal_contract COMMAND scene_signal_tests "$<TARGET_FILE:scene_signal_test_receiver>" "$<TARGET_FILE:scene_signal_missing_export>")
if(MSVC)
    target_compile_options(scene_signal_tests PRIVATE /W4 /WX /EHsc)
    target_compile_options(scene_signal_test_receiver PRIVATE /W4 /WX /EHsc)
endif()
